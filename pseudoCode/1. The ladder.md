START // Begin The Ladder program

FUNCTION playTheLadder
    VARIABLE totalThrows = 0
    VARIABLE currentTarget = 1

    PRINT "Starting the game: The Ladder! Goal is to roll 1 to 6 in order."

    WHILE currentTarget LESS THAN OR EQUAL TO 6
        VARIABLE dieRoll = Random number between 1 and 6
        SET totalThrows = totalThrows + 1
        
        PRINT "Throw #" + totalThrows + ": Rolled " + dieRoll + ". Looking for " + currentTarget + "."

        IF dieRoll IS EQUAL TO currentTarget THEN
            PRINT "Success! You found " + currentTarget + "."
            SET currentTarget = currentTarget + 1
        ENDIF
    ENDWHILE

    PRINT "Game Over! You completed the ladder in " + totalThrows + " throws."
ENDFUNCTION

END // End program
