START // Begin Knock Out program

FUNCTION playKnockOut
    INPUT knockOutNumber
    
    // Validate choice (must be 6, 7, 8, or 9)
    IF knockOutNumber IS NOT 6 AND NOT 7 AND NOT 8 AND NOT 9 THEN
        PRINT "Invalid choice. Please pick 6, 7, 8, or 9."
        EXIT FUNCTION
    ENDIF

    VARIABLE totalScore = 0
    VARIABLE isKnockedOut = FALSE

    WHILE isKnockedOut IS FALSE
        VARIABLE dieOne = Random number between 1 and 6
        VARIABLE dieTwo = Random number between 1 and 6
        VARIABLE currentSum = dieOne + dieTwo

        PRINT "Rolled " + dieOne + " and " + dieTwo + ". Sum is " + currentSum

        IF currentSum IS EQUAL TO knockOutNumber THEN
            PRINT "Knocked out! You hit your knockout number: " + knockOutNumber
            SET isKnockedOut = TRUE
        ELSE
            SET totalScore = totalScore + currentSum
            PRINT "Safe! Current score: " + totalScore
        ENDIF
    ENDWHILE

    PRINT "Game Over! Your final score is: " + totalScore
ENDFUNCTION

END // End program
