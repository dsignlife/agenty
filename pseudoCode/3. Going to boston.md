START // Begin Going to Boston program

FUNCTION playGoingToBoston
    VARIABLE totalScore = 0
    VARIABLE remainingDice = 3

    WHILE remainingDice GREATER THAN 0
        VARIABLE currentRolls = Empty List

        // Roll remaining dice
        FOR count FROM 1 TO remainingDice
            VARIABLE dieRoll = Random number between 1 and 6
            ADD dieRoll TO currentRolls
        ENDFOR

        // Find and keep the highest die
        VARIABLE highestDie = Highest value in currentRolls
        SET totalScore = totalScore + highestDie

        PRINT "Rolled: " + currentRolls + ". Kept highest: " + highestDie
        PRINT "Running Total: " + totalScore

        // Reduce dice count for next round
        SET remainingDice = remainingDice - 1
    ENDWHILE

    PRINT "Final score for Going to Boston: " + totalScore
ENDFUNCTION

END // End program
