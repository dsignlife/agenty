START // Begin Rock, Paper, Scissors program

FUNCTION playRockPaperScissors
    INPUT playerChoice
    VARIABLE validChoices = ["rock", "paper", "scissors"]

    IF playerChoice IS NOT IN validChoices THEN
        PRINT "Invalid input! Please choose rock, paper, or scissors."
        EXIT FUNCTION
    ENDIF

    VARIABLE computerChoice = Random selection from validChoices

    PRINT "Player chose: " + playerChoice
    PRINT "Computer chose: " + computerChoice

    IF playerChoice IS EQUAL TO computerChoice THEN
        PRINT "Result: It's a draw!"
    ELSE IF playerChoice IS "rock" AND computerChoice IS "scissors" THEN
        PRINT "Result: Player wins!"
    ELSE IF playerChoice IS "paper" AND computerChoice IS "rock" THEN
        PRINT "Result: Player wins!"
    ELSE IF playerChoice IS "scissors" AND computerChoice IS "paper" THEN
        PRINT "Result: Player wins!"
    ELSE
        PRINT "Result: Computer wins!"
    ENDIF
ENDFUNCTION

END // End program
