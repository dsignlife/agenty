START // Begin FizzBuzz program

FUNCTION playFizzBuzz
    FOR currentNumber FROM 1 TO 100
        IF (currentNumber MOD 3 IS 0) AND (currentNumber MOD 5 IS 0) THEN
            PRINT "FizzBuzz"
        ELSE IF currentNumber MOD 3 IS 0 THEN
            PRINT "Fizz"
        ELSE IF currentNumber MOD 5 IS 0 THEN
            PRINT "Buzz"
        ELSE
            PRINT currentNumber
        ENDIF
    ENDFOR
ENDFUNCTION

END // End program
