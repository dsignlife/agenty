
START // Begin High or Low program

FUNCTION playHighOrLow
    VARIABLE secretNumber = Random number between 1 and 100
    VARIABLE hasGuessedCorrectly = FALSE
    VARIABLE userGuess = 0

    PRINT "Welcome to High or Low! Try to guess the secret number."

    WHILE hasGuessedCorrectly IS FALSE
        PRINT "Enter a number between 1 and 100:"
        INPUT userGuess

        IF userGuess IS EQUAL TO secretNumber THEN
            PRINT "Congratulations! You guessed the secret number: " + secretNumber
            SET hasGuessedCorrectly = TRUE
        ELSE IF userGuess LESS THAN secretNumber THEN
            PRINT "Too low! Try a higher number."
        ELSE IF userGuess GREATER THAN secretNumber THEN
            PRINT "Too high! Try a lower number."
        ENDIF
    ENDWHILE
ENDFUNCTION

END // End program
