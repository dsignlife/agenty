START // Begin Todo App program

VARIABLE todoList = Empty List
VARIABLE nextTaskId = 1

FUNCTION addTask
    INPUT taskDescription
    VARIABLE newTask = Object with (id: nextTaskId, description: taskDescription, isCompleted: FALSE)
    
    ADD newTask TO todoList
    SET nextTaskId = nextTaskId + 1
    PRINT "Added task: " + taskDescription
ENDFUNCTION

FUNCTION toggleTaskStatus
    INPUT targetTaskId
    FIND task IN todoList WHERE task.id IS EQUAL TO targetTaskId

    IF task EXISTS THEN
        SET task.isCompleted = NOT task.isCompleted
        PRINT "Status updated for task ID: " + targetTaskId
    ELSE
        PRINT "Task not found."
    ENDIF
ENDFUNCTION

FUNCTION removeTask
    INPUT targetTaskId
    FIND taskIndex IN todoList WHERE task.id IS EQUAL TO targetTaskId

    IF taskIndex EXISTS THEN
        REMOVE item at taskIndex FROM todoList
        PRINT "Removed task ID: " + targetTaskId
    ELSE
        PRINT "Task not found."
    ENDIF
ENDFUNCTION

FUNCTION showTodoList
    PRINT "--- My Todo List ---"
    IF todoList IS EMPTY THEN
        PRINT "Your todo list is empty!"
    ELSE
        FOR EACH task IN todoList
            VARIABLE checkmark = IF task.isCompleted THEN "[X]" ELSE "[ ]"
            PRINT task.id + ". " + checkmark + " " + task.description
        ENDFOR
    ENDIF
ENDFUNCTION

END // End program
